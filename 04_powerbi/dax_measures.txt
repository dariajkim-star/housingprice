/* ===========================================================
   과제 4·5 Power BI — DAX 모음
   시각화 색 규칙 (과제 1·2·3 파이썬 차트와 동일)
     남색 #1F4E79 : 핵심 강조
     회색 #C8CDD3 : 일반 데이터
     빨강 #B94A48 : 이상치·주의
   =========================================================== */


/* -----------------------------------------------------------
   1) 기본 측정값
   ----------------------------------------------------------- */

평균 SalePrice = AVERAGE ( 'data'[SalePrice] )

주택 수 = COUNTROWS ( 'data' )

최고 SalePrice = MAX ( 'data'[SalePrice] )

전체 평균 SalePrice = CALCULATE ( AVERAGE ( 'data'[SalePrice] ), ALL ( 'data' ) )


/* -----------------------------------------------------------
   2) 계산 열 — 과제 5 Key Influencers 용
   ----------------------------------------------------------- */

SalePrice_등급 =
IF (
    'data'[SalePrice] >= MEDIANX ( ALL ( 'data' ), 'data'[SalePrice] ),
    "High",
    "Low"
)


/* -----------------------------------------------------------
   3) 지역별 막대 — 조건부 색상 측정값

   적용법: 막대 차트 선택 → 서식(붓) → 막대 → 색 →
           오른쪽 fx 아이콘 클릭 → 서식 스타일 = "필드 값" →
           필드에 아래 측정값 선택

   [A안] 상위 3개 지역만 강조  ← 메시지가 가장 선명
   ----------------------------------------------------------- */

지역색_상위3 =
VAR 지역평균 = AVERAGE ( 'data'[SalePrice] )
VAR 순위 =
    RANKX (
        ALL ( 'data'[Neighborhood] ),
        CALCULATE ( AVERAGE ( 'data'[SalePrice] ) ),
        ,
        DESC
    )
RETURN
    IF ( 순위 <= 3, "#1F4E79", "#C8CDD3" )


/* [B안] 전체 평균 이상 + 표본 20건 이상만 강조
         ← 과제 1 ④번 파이썬 차트와 동일한 규칙 (표본 적은 지역 제외)   */

지역색_평균이상 =
VAR 지역평균 = AVERAGE ( 'data'[SalePrice] )
VAR 전체평균 = CALCULATE ( AVERAGE ( 'data'[SalePrice] ), ALL ( 'data' ) )
VAR 표본수 = COUNTROWS ( 'data' )
RETURN
    IF ( 지역평균 >= 전체평균 && 표본수 >= 20, "#1F4E79", "#C8CDD3" )


/* -----------------------------------------------------------
   4) 건축연도 꺾은선 — 최근 구간 강조용 (선택)
   ----------------------------------------------------------- */

연도색_2000년대 =
IF ( SELECTEDVALUE ( 'data'[YearBuilt] ) >= 2000, "#1F4E79", "#C8CDD3" )


/* -----------------------------------------------------------
   5) 산점도 이상치 표시 (선택)
      과제 1 ③에서 확인한 Id 524·1299 = SaleCondition Partial 미완성 거래
   ----------------------------------------------------------- */

산점도색 =
VAR 면적 = SELECTEDVALUE ( 'data'[GrLivArea] )
VAR 가격 = SELECTEDVALUE ( 'data'[SalePrice] )
RETURN
    IF ( 면적 > 4000 && 가격 < 300000, "#B94A48", "#7B8794" )
